#include <stdio.h>

int max(int a, int b)
{
    return (a > b) ? a : b;
}

int main()
{
    int n, capacity;
    int value[100], weight[100];
    int dp[100][100];

    scanf("%d", &n);

    for (int i = 0; i < n; i++)
        scanf("%d", &value[i]);

    for (int i = 0; i < n; i++)
        scanf("%d", &weight[i]);

    scanf("%d", &capacity);

    // Initialize DP table
    for (int i = 0; i <= n; i++)
    {
        for (int w = 0; w <= capacity; w++)
        {
            if (i == 0 || w == 0)
            {
                dp[i][w] = 0;
            }
            else if (weight[i - 1] <= w)
            {
                dp[i][w] = max(
                    value[i - 1] + dp[i - 1][w - weight[i - 1]],
                    dp[i - 1][w]
                );
            }
            else
            {
                dp[i][w] = dp[i - 1][w];
            }
        }
    }

    printf("%d", dp[n][capacity]);

    return 0;
}
